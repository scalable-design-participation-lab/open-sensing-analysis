def is_inside_interval(interval, date):
    start, end = interval
    crosses_new_year = (start.month, start.day) > (end.month, end.day)
    if crosses_new_year:
        return start.is_bigger_or_equal_than(date) or end.is_less_or_equal_than(date)

    return start.is_bigger_or_equal_than(date) and end.is_less_or_equal_than(date)

#Find indices common to both arrays
def get_included_indices(reference_values, tested_values):
    n = len(tested_values)
    included_indices = []
    filtered_values = []

    for i in range(0,n):
        tested_value = tested_values[i]
        if(tested_value in reference_values):
            filtered_values.append(tested_value)
            included_indices.append(i)
        
    
    return included_indices, filtered_values

def calculate_differences(season_test_dictionary, season_ref_dictionary):
    #Timestamps are the indices
    reference_timestamps = set(season_ref_dictionary["df"].index)
    analyzed_sensor_timestamps = list(season_test_dictionary["df"].index)
    #Check the time overlap
    included_indices, filtered_values = get_included_indices(
        reference_values = reference_timestamps,
        tested_values = analyzed_sensor_timestamps
    )

    all_temperature_values = list(season_test_dictionary["df"]["temperature"])
    temperature_differences = []
    for i in range(0,len(included_indices)):
        index = included_indices[i]
        timestamp = analyzed_sensor_timestamps[index]
        reference_temperature = season_ref_dictionary["df"].loc[timestamp, "temperature"]
        temperature = all_temperature_values[index]
        temperature_delta = temperature - reference_temperature
        temperature_differences.append(temperature_delta)
    
    season_test_dictionary["temperature_diffs"] = temperature_differences
    season_test_dictionary["diff_timestamps"] = filtered_values

    shadow_values_0m = list(season_test_dictionary["shadows"]["shadow_radius_0m"])
    filtered_shadows = [shadow_values_0m[index] for index in included_indices]
    season_test_dictionary["filtered_shadows_0m"] = filtered_shadows

def split_day_and_night(
        sensor_dictionary
    ):

    model_df = sensor_dictionary["shadows"].loc[sensor_dictionary["diff_timestamps"], shadow_columns]
    model_df["temperature_diff"] = sensor_dictionary["temperature_diffs"]
    model_df = model_df.dropna(subset = ["temperature_diff"])

    is_day = model_df["shadow_radius_0m"].notna()
    sensor_dictionary["day_hours"] = model_df[is_day]
    sensor_dictionary["night_hours"] = model_df[~is_day]

    hour_counts= {
        "day hours in the sun": int((model_df["shadow_radius_0m"] == 1).sum()),
        "day hours in the shadow": int((model_df["shadow_radius_0m"] == 0).sum()),
        "night hours": int((~is_day).sum())
    }

    sensor_dictionary["hour_counts"] = hour_counts

def get_models_for_dictionary(data_dictionary):
    models_per_radius = []
    day_hours_df = data_dictionary["day_hours"]
    for radius in MODEL_RADII:
        day_hours_at_radius = day_hours_df.assign(
            shadow_value = day_hours_df[f"shadow_radius_{radius}m"]
        )
        model = smf.ols("temperature_diff ~ shadow_value", data = day_hours_at_radius).fit(
                cov_type = "cluster", 
                cov_kwds = {"groups": day_hours_at_radius["date"]}
            )
        
        models_per_radius.append(model)
    
    return models_per_radius
